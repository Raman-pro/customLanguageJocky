Fn is_prime -> bool (n:int)
  If
    Binary(<)
      Ident(n)
      IntLit(2)
    Then:
      Return        BoolLit(false)

  Let i : int =    IntLit(2)

  While
    Binary(<=)
      Binary(*)
        Ident(i)
        Ident(i)
      Ident(n)
    If
      Binary(==)
        Binary(%)
          Ident(n)
          Ident(i)
        IntLit(0)
      Then:
        Return          BoolLit(false)

    ExprStmt
      Binary(=)
        Ident(i)
        Binary(+)
          Ident(i)
          IntLit(1)
  Return    BoolLit(true)

Fn main -> void ()
